package es.uab.tqs.f1_2D.controlador;

import es.uab.tqs.f1_2D.model.AICar;
import es.uab.tqs.f1_2D.model.Car;
import es.uab.tqs.f1_2D.model.Map;

import java.awt.image.BufferedImage;
import java.awt.Point;
import java.util.*;

public class AIController 
{
    private List<AICar> aiCars;
    private Map trackMap;
    private List<Point> racingLine;
    private Car playerCar;
    
    // Equipos F1 con colores característicos
    private final String[] F1_TEAMS = 
    {
        "Mercedes", "Red Bull", "Ferrari", "McLaren", "Alpine", 
        "Racing Bulls", "Aston Martin", "Williams", "Sauber", "Haas"
    };
    
    public AIController(Map trackMap, Car playerCar) 
    {
        this.trackMap = trackMap;
        this.playerCar = playerCar;
        this.aiCars = new ArrayList<>();
        this.racingLine = new ArrayList<>();
        
        generateRacingLine();
        initializeAICars();
    }
    
    private void generateRacingLine() 
{
    racingLine.clear(); // Limpiar primero
    
    // Waypoints principales con interpolación
    Point[] keyPoints = {
        new Point(431, 1166), 
        new Point(600, 823), 
        new Point(817, 585), 
        new Point(1240, 668), 
        new Point(1728, 760), 
        new Point(2472, 872), 
        new Point(3152, 976), 
        new Point(3888, 988), 
        new Point(4548, 1056), 
        new Point(4816, 928), 
        new Point(4908, 780), 
        new Point(4860, 632), 
        new Point(4762, 462), 
        new Point(4888, 303), 
        new Point(5032, 252), 
        new Point(5324, 196), 
        new Point(5632, 136), 
        new Point(5752, 208), 
        new Point(5752, 372), 
        
        // HORQUILLA MEJORADA - Puntos de entrada, vértice y salida
        new Point(5800, 520),  // Punto de frenada - ANTES de la curva
        new Point(5868, 569),  // Entrada de curva
        new Point(5920, 592),  // Punto intermedio 1
        new Point(5970, 598),  // Punto intermedio 2  
        new Point(6028, 585),  // Vértice de la curva
        new Point(6059, 540),  // Punto intermedio 3
        new Point(6078, 472),  // Punto intermedio 4
        new Point(6079, 386),  // Punto intermedio 5
        new Point(6113, 350),  // Punto intermedio 6
        new Point(6227, 324),  // sALIDA      
        new Point(6296, 332), 
        new Point(6436, 444), 
        new Point(6452, 664), 
        new Point(6408, 840), 
        new Point(6196, 1020), 
        new Point(5808, 1180), 
        new Point(5032, 1236), 
        new Point(3648, 1200), 
        new Point(3420, 1264), 
        new Point(3292, 1332), 
        new Point(2976, 1268), 
        new Point(2128, 1040), 
        new Point(1856, 1024), 
        new Point(1492, 1220), 
        new Point(1644, 1920), 
        new Point(1412, 2064), 
        new Point(1492, 2288), 
        new Point(1900, 2468), 
        new Point(1880, 2636), 
        new Point(1496, 2716), 
        new Point(760, 2708), 
        new Point(640, 2616), 
        new Point(472, 2232), 
        new Point(344, 1744), 
        new Point(360, 1480), 
        new Point(444, 1344)
    };
    
    // Añadir waypoints intermedios para mejor navegación
    for (int i = 0; i < keyPoints.length - 1; i++) {
        Point current = keyPoints[i];
        Point next = keyPoints[i + 1];
        
        // Añadir punto actual
        racingLine.add(current);
        
        // Añadir puntos intermedios si la distancia es grande
        double distance = Math.sqrt(Math.pow(next.x - current.x, 2) + Math.pow(next.y - current.y, 2));
        if (distance > 150) {
            int intermediatePoints = (int)(distance / 75); // Punto cada ~100 pixels
            for (int j = 1; j < intermediatePoints; j++) {
                double t = (double)j / intermediatePoints;
                int interX = (int)(current.x + (next.x - current.x) * t);
                int interY = (int)(current.y + (next.y - current.y) * t);
                racingLine.add(new Point(interX, interY));
            }
        }
    }
    
    // Añadir el último punto
    racingLine.add(keyPoints[keyPoints.length - 1]);
    
}
    
    private void initializeAICars() 
    {
        Random random = new Random();

        int numberOfCars = 19;
        
        // Listas personalizables de posiciones X e Y para la parrilla de salida
        double[] startPositionsX = {
            444, 
            360, 428, 
            336, 420, 
            332, 424, 
            324, 432, 
            340, 444, 
            356, 464,
            384, 496, 
            412, 496, 
            436, 502
        };
        
        double[] startPositionsY = {
            1344, 
            1376, 1448, 
            1476, 1544, 
            1572, 1644, 
            1676, 1736, 
            1776, 1836, 
            1872, 1936, 
            1976, 2036, 
            2076, 2132, 
            2176, 2194
        };
        
        for (int i = 0; i < numberOfCars; i++) { 
            String team = F1_TEAMS[i % F1_TEAMS.length];
            
            double skillLevel;
 
            if (i < 5) skillLevel = 0.7 + random.nextDouble() * 0.3; 
            else if (i < 13) skillLevel = 0.4 + random.nextDouble() * 0.3; 
            else skillLevel = 0.2 + random.nextDouble() * 0.3; 

            // Usar las posiciones definidas en las listas
            double startX = startPositionsX[i];
            double startY = startPositionsY[i];
            
            // Calcular ángulo hacia el primer waypoint
            double startAngle = calculateAngleToWaypoint(startX, startY, 1);

            double baseMaxVelocity = 22 + (random.nextDouble() * 4);
            
            AICar aiCar = new AICar(
                startX, startY, startAngle, 3.0, 
                baseMaxVelocity, -5, 0.2, 0.5,
                trackMap.getMapHeight(), trackMap.getMapWidth(),
                team, skillLevel
            );

            aiCar.setRacingLine(new ArrayList<>(racingLine));
            aiCar.setPlayerCar(playerCar);
            aiCars.add(aiCar);
        }
    }

    private double calculateAngleToWaypoint(double startX, double startY, int waypointIndex) 
    {
        if (racingLine.size() <= waypointIndex) {
            return 280; // Ángulo por defecto
        }
        
        Point waypoint = racingLine.get(waypointIndex);
        double dx = waypoint.x - startX;
        double dy = waypoint.y - startY;
        double angle = Math.toDegrees(Math.atan2(dy, dx));
        
        // Asegurar que el ángulo esté en el rango correcto
        if (angle < 0) {
            angle += 360;
        }
        return angle;
    }
    
    public void updateAllAI(BufferedImage collisionMap) 
    {
        int currentLap = trackMap.getCurrentLap();
        
        for (AICar aiCar : aiCars)
        {
            aiCar.updateAI(collisionMap, aiCars, currentLap);
        }
    }
    
    public List<AICar> getAICars() {return aiCars;}
    public List<Point> getRacingLine() {return racingLine;}
    
    public void setAIDifficulty(double globalDifficulty) 
    {
        for (AICar aiCar : aiCars) 
        {
            double currentSkill = aiCar.getSkillLevel();
            double newSkill = currentSkill * globalDifficulty;
            aiCar.setSkillLevel(Math.min(1.0, newSkill));
        }
    }
}
